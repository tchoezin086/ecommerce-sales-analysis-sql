Drop table orders;
CREATE TABLE products (
    product_id INT PRIMARY KEY,
    product_name VARCHAR(100),
    category VARCHAR(50),
    price DECIMAL(10,2)
);
CREATE TABLE orders (
    order_id INT PRIMARY KEY,
    customer_id INT,
    order_date DATE,
    payment_mode VARCHAR(30),
    order_status VARCHAR(20),
    FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id)
);
CREATE TABLE order_items (
    order_item_id INT PRIMARY KEY,
    order_id INT,
    product_id INT,
    quantity INT,
    unit_price DECIMAL(10,2),
    discount DECIMAL(5,2),
    FOREIGN KEY (order_id)
        REFERENCES orders(order_id),
    FOREIGN KEY (product_id)
        REFERENCES products(product_id)
);

-- Level 1 — SQL basics
-- Display all customers from Delhi.
SELECT * FROM customers WHERE city="Delhi";
-- Find all products priced above ₹10,000.
SELECT * FROM products WHERE price>10000;
-- Display all delivered orders.
SELECT *FROM orders WHERE order_status="Delivered";
-- Find products in the Electronics category.
SELECT * FROM products WHERE category="Electronics";
-- Sort products from highest to lowest price.
SELECT * FROM  products ORDER BY price DESC;
-- Level 2 — Aggregation
--  Count the total number of customers.
SELECT COUNT(customer_id) from customers;
-- Calculate the average product price.
SELECT avg(price) from products;
-- Find the minimum and maximum product price.
SELECT min(price) FROM products;
SELECT max(price) FROM products;
-- Count orders by order status.
SELECT count(order_status) FROM orders;
-- Find the total quantity sold for each product.
SELECT p.product_name,sum(oi.quantity) AS Total_Quantity_Sold FROM order_items oi INNER JOIN products p ON oi.product_id=p.product_id GROUP BY p.product_name;
-- Level 3 — Business analysis
-- Calculate total sales revenue from delivered orders, after discounts.
SELECT p.category,Round
  (sum(
      oi.quantity*oi.unit_price
      *(1-oi.discount/100)
), 2
) AS "Total_Revenue" 
FROM order_items oi
INNER JOIN  products p ON p.product_id=oi.product_id
INNER JOIN orders o ON oi.order_id=o.order_id
WHERE order_status="Delivered"
GROUP BY p.category
ORDER BY Total_Revenue DESC;
-- Find total revenue by product category.
SELECT p.category,
SUM(oi.quantity*oi.unit_price
*(1-discount/100)) AS Total_Revenue 
FROM order_items oi
INNER JOIN products p on p.product_id=oi.product_id
INNER JOIN orders o on oi.order_id=o.order_id
WHERE order_status="Delivered"
GROUP BY p.category
ORDER BY Total_Revenue DESC;
-- Identify the top 3 products by revenue.
SELECT p.category,
SUM(oi.quantity*oi.unit_price
*(1-discount/100)) AS Total_Revenue 
FROM order_items oi
INNER JOIN products p on p.product_id=oi.product_id
INNER JOIN orders o on oi.order_id=o.order_id
WHERE order_status="Delivered"
GROUP BY p.category
ORDER BY Total_Revenue DESC
LIMIT 3;
-- Calculate total revenue by city.
SELECT c.city,
SUM(oi.quantity*oi.unit_price
*(1-discount/100)) AS Total_Revenue 
FROM customers c 
INNER JOIN orders o on c.customer_id=o.customer_id
INNER JOIN order_items oi on o.order_id=oi.order_id
WHERE order_status="Delivered"
GROUP BY c.city
ORDER BY Total_Revenue DESC
-- Find customers who have placed more than one order.
SELECT c.customer_name,
count(o.order_id) AS "total_order"
FROM customers c 
INNER JOIN orders o 
on c.customer_id=o.customer_id
GROUP BY c.customer_id,c.customer_name
HAVING COUNT(o.order_id) > 1;
-- Level 4 — JOINs and advanced SQL
-- Display each order with the customer name and order date.
SELECT
    o.order_id,
    c.customer_name,
    o.order_date
FROM customers c
INNER JOIN orders o
    ON c.customer_id = o.customer_id;
--  Find products that have never been ordered.
SELECT p.product_id,p.product_name 
FROM products p 
LEFT JOIN order_items oi
 on oi.product_id=p.product_id
 WHERE oi.product_id IS Null;
--  Find the highest-spending customer.
SELECT c.customer_id,c.customer_name,
ROUND(SUM(oi.quantity*oi.unit_price*(1-discount/100)),2) AS "Total_Spending"
FROM customers c
INNER JOIN orders o
ON o.customer_id=c.customer_id
INNER JOIN order_items oi
ON oi.order_id=o.order_id
GROUP BY c.customer_id,c.customer_name
ORDER BY Total_Spending DESC
LIMIT 1;

-- Compare cancelled, pending, and delivered order counts.
SELECT order_status,
COUNT(order_id) 
as Total_orders 
FROM orders 
GROUP BY order_status 
ORDER BY Total_orders DESC;
-- Rank products by revenue using a window function.
SELECT
    product_name,
    total_revenue,
    RANK() OVER (ORDER BY total_revenue DESC) AS revenue_rank
FROM (
    SELECT
        p.product_name,
        SUM(oi.quantity * oi.unit_price * (1 - oi.discount / 100)) AS total_revenue
    FROM products p
    INNER JOIN order_items oi
        ON p.product_id = oi.product_id
    INNER JOIN orders o
        ON oi.order_id = o.order_id
    WHERE o.order_status = 'Delivered'
    GROUP BY p.product_id, p.product_name
) AS product_revenue
ORDER BY revenue_rank;

-- Level 5 — Analyst insights
-- Identify the best-performing month by revenue.
SELECT DATE_FORMAT(o.order_date, '%m') AS month,
ROUND(sum(oi.quantity*oi.unit_price*(1-oi.discount/100)),2) AS Total_Revenue 
FROM orders o
INNER JOIN order_items oi ON oi.order_id=o.order_id
WHERE o.order_status = 'Delivered'
GROUP BY month
ORDER BY Total_Revenue DESC
LIMIT 1;
-- Find the most popular payment mode.
SELECT payment_mode,count(*) AS total_orders FROM orders
GROUP BY payment_mode
ORDER BY total_orders DESC
LIMIT 1;
-- Calculate each category's percentage contribution to total revenue.
SELECT p.category,
ROUND(SUM(oi.quantity*oi.unit_price*(1-oi.discount/100)),2) 
AS Total_Revenue,
ROUND(SUM(oi.quantity*oi.unit_price*(1-oi.discount/100))
/(SELECT SUM(quantity*unit_price*(1-discount/100))
 FROM order_items)*100,2) AS Revenue_Percentage
FROM products p 
INNER JOIN order_items oi
ON oi.product_id=p.product_id
GROUP BY p.category
ORDER BY Total_Revenue DESC;
-- Segment customers into high, medium, and low spenders using CASE WHEN.
SELECT c.customer_id,c.customer_name,
ROUND(SUM(oi.quantity*oi.unit_price*(1-oi.discount/100)),2) 
AS Total_Spending,
CASE  -- WE CAN ALSO USE IF.The main difference is that IF() handles one condition at a time, while CASE WHEN is easier to read when you have multiple categories.
  WHEN SUM(oi.quantity*oi.unit_price*(1-oi.discount/100))>=50000
    THEN "High_Spender"
  WHEN SUM(oi.quantity*oi.unit_price*(1-oi.discount/100))>=20000
    THEN "Medium_Spender"
  ELSE "Low_Spender"
END AS Spending_Segemnt
FROM customers c
INNER JOIN orders o ON c.customer_id=o.customer_id
INNER JOIN order_items oi ON oi.order_id=o.order_id
WHERE o.order_status="Delivered"
GROUP BY c.customer_id,c.customer_name
ORDER BY Total_Spending DESC;
